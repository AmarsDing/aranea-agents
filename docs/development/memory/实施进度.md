  option (google.api.http) = {get: "/v1/memory/layer-overview"};

rpc GetUnifiedMemoryGraph(GetUnifiedMemoryGraphRequest) returns (GetUnifiedMemoryGraphResponse) {
  option (google.api.http) = {get: "/v1/memory/graph/unified"};

message GetMemoryLayerOverviewRequest { string agent_id = 1; string session_id = 2; }
message MemoryLayerStat {
  string layer = 1;              // L0..L4
  int32 item_count = 2;
  int32 today_added = 3;
  int32 recall_hits = 4;

message MemoryActionItem { string kind = 1; int32 count = 2; string target_tab = 3; }

}
message GetMemoryLayerOverviewResponse {
  repeated MemoryLayerStat layers = 1;
  repeated MemoryActionItem action_items = 2;
  repeated MemoryActivityItem activity_feed = 3;
}

message GetUnifiedMemoryGraphRequest {
  string agent_id = 1; string focus = 2; int32 hops = 3;
  double min_weight = 4; repeated string layers = 5;
}
message UnifiedGraphNode {
  string id = 1; string layer = 2; string kind = 3; string label = 4;
  double weight = 5; string meta_json = 6;
}
message UnifiedGraphEdge {
  string source = 1; string target = 2; string type = 3;
  string label = 4; double weight = 5; string polarity = 6;
}
message GetUnifiedMemoryGraphResponse {
  string focus = 1;
  repeated UnifiedGraphNode nodes = 2;
  repeated UnifiedGraphEdge edges = 3;
  int32 node_count = 4; int32 edge_count = 5; int32 filtered_edge_count = 6;

}
