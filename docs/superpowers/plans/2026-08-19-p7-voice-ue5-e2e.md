```python
# test/voice-ue-e2e/t1_smoke_latency.py
"""
T1 冒烟：KWS→ASR→Spirit→TTS 全链路延迟 <2s。
手段：经 Voice Bridge text-command 调试端点注入文本（等价于 ASR final 之后的管线），
测量从提交到收到 say 帧的往返延迟；同时校验 WS 链路 state 帧流转。
"""
import json, os, time, requests, websocket

VOICE_HTTP = os.getenv("VOICE_HTTP", "http://127.0.0.1:8140")
VOICE_WS = os.getenv("VOICE_WS", "ws://127.0.0.1:8140/api/v1/monitor/voice/ws")
JWT = os.getenv("VOICE_JWT", "")  # dev bypass 可空

def test_http_text_command_latency():
    t0 = time.monotonic()
    r = requests.post(
        f"{VOICE_HTTP}/api/v1/monitor/voice/text-command",
        headers={"Authorization": f"Bearer {JWT}", "Content-Type": "application/json"},
        json={"text": "查看机柜 A12", "channel": "web"},
        timeout=10,
    )
    dt = time.monotonic() - t0
    assert r.status_code == 200, f"text-command status={r.status_code}"
    body = r.json()
    # 契约：{say, scene_actions, trace_id}
    assert "say" in body and "trace_id" in body, f"bad contract: {body}"
    print(f"[T1] text-command latency={dt:.3f}s say={body['say'][:50]}")
    assert dt < 2.0, f"latency {dt:.3f}s >= 2.0s"

def test_ws_state_frames():
    """WS 链路：auth → state 帧（idle/listening/thinking/speaking）依次下行。"""
    ws = websocket.create_connection(f"{VOICE_WS}?channel=web", timeout=5)
    ws.send(json.dumps({"type": "auth", "token": JWT}))
    frames = []
    t0 = time.monotonic()
    while time.monotonic() - t0 < 5:
        raw = ws.recv()
        if isinstance(raw, bytes):
            continue  # TTS 二进制帧跳过
        msg = json.loads(raw)
        frames.append(msg)
        if msg.get("type") == "state" and msg.get("state") == "idle":
            break
    ws.close()
    types = [f.get("type") for f in frames]
    print(f"[T1] ws frames={types}")
    assert "state" in types, f"no state frame: {frames}"

if __name__ == "__main__":
    test_http_text_command_latency()
    test_ws_state_frames()
    print("T1 PASS")
```

```python
# test/voice-ue-e2e/t3_coreference.py
"""
T3：L0 会话记忆（聚焦实体栈）指代消解。
机制：text_cmd 上行帧携带 focus_stack（客户端回传），Voice Bridge 提交 aranea 时经 [CONTEXT] 块注入；
      aranea Spirit 依据栈顶实体消解"它/这个/刚才的"等指代。
断言：第二轮 scene_action 的 entity.id 解析为 A12 或其内服务器，而非空/其他机柜。
"""
import json, os, time, requests, websocket

VOICE_HTTP = os.getenv("VOICE_HTTP", "http://127.0.0.1:8140")
VOICE_WS = os.getenv("VOICE_WS", "ws://127.0.0.1:8140/api/v1/monitor/voice/ws")
JWT = os.getenv("VOICE_JWT", "")

def _round(ws, text, focus_stack):
    ws.send(json.dumps({"type": "text_cmd", "text": text, "focus_stack": focus_stack}))
    actions, say = [], ""
    t0 = time.monotonic()
    while time.monotonic() - t0 < 10:
        raw = ws.recv()
        if isinstance(raw, bytes):
            continue
        msg = json.loads(raw)
        if msg.get("type") == "scene_action":
            actions.append(msg)
        elif msg.get("type") == "say":
            say = msg.get("text", "")
        elif msg.get("type") == "state" and msg.get("state") == "idle":
            break
    return actions, say

def test_coreference_via_focus_stack():
    ws = websocket.create_connection(f"{VOICE_WS}?channel=web", timeout=8)
    ws.send(json.dumps({"type": "auth", "token": JWT}))
    time.sleep(0.5)

    # Round 1：聚焦 A12
    a1, s1 = _round(ws, "查看 A12 机柜", [])
    stack = [{"type": "cabinet", "id": "A12"}]

    # Round 2：指代消解 —— "它的 CPU 呢" 应解析到 A12 上下文
    a2, s2 = _round(ws, "它的 CPU 呢", stack)
    print(f"[T3] round2 actions={a2} say={s2}")
    ws.close()

    # 断言：第二轮产出聚焦/详情类指令，且目标在 A12 上下文（server 或 part）
    assert a2, f"round2 no scene_action: say={s2}"
    targets = [a.get("entity", {}) for a in a2 if a.get("entity")]
    # 宽容断言：entity.id 含 A12 关联（如 SV-03@A12 或直接 A12），或 say 提及 A12/CPU
    joined = json.dumps(targets, ensure_ascii=False) + s2
    assert ("A12" in joined) or ("CPU" in joined), f"coreference lost: {joined}"

    """GET /sessions/{id} 返回聚焦栈（管理页可观测）。"""

    detail = r2.json()
    print(f"[T3] session detail={detail}")
    assert "focus_stack" in json.dumps(detail), f"no focus_stack: {detail}"

if __name__ == "__main__":
    test_coreference_via_focus_stack()
    test_session_focus_stack_visible()
    print("T3 PASS")
```
